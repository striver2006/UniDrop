+def in_round_rect(px, py, x0, y0, x1, y1, r):

+    return dx * dx + dy * dy <= r * r

+    raw = bytearray()

+        + chunk(b"IHDR", ihdr)
+        + chunk(b"IDAT", zlib.compress(bytes(raw), 9))
+        + chunk(b"IEND", b"")
+    )
+    path.write_bytes(png)
